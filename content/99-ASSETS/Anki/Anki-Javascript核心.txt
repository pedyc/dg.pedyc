手写一个简易版 Promise（实现 then、resolve、reject 核心逻辑）。	class MyPromise {<br>  constructor(executor) {<br>    this.state = 'pending';<br>    this.value = undefined;<br>    this.callbacks = [];<br>    const resolve = (val) => {<br>      if (this.state !== 'pending') return;<br>      this.state = 'fulfilled';<br>      this.value = val;<br>      this.callbacks.forEach(cb => cb.onFulfilled(val));<br>    };<br>    const reject = (err) => {<br>      if (this.state !== 'pending') return;<br>      this.state = 'rejected';<br>      this.value = err;<br>      this.callbacks.forEach(cb => cb.onRejected(err));<br>    };<br>    try { executor(resolve, reject); } catch (e) { reject(e); }<br>  }<br>  then(onFulfilled, onRejected) {<br>    return new MyPromise((resolve, reject) => {<br>      const handle = () => {<br>        try {<br>          const cb = this.state === 'fulfilled' ? onFulfilled : onRejected;<br>          if (!cb) {<br>            (this.state === 'fulfilled' ? resolve : reject)(this.value);<br>            return;<br>          }<br>          const result = cb(this.value);<br>          resolve(result);<br>        } catch (err) {<br>          reject(err);<br>        }<br>      };<br>      if (this.state === 'pending') {<br>        this.callbacks.push({ onFulfilled: handle, onRejected: handle });<br>      } else {<br>        setTimeout(handle, 0);<br>      }<br>    });<br>  }<br>}<br><br>注：含核心流程，未实现链式传递中 then 返回 Promise 的完整规范。	JavaScript 手写
